const controlIds = ['fl', 'fr', 'rl', 'rr'];

export function initializeSuspensaoSimulation(module, root) {
  const controls = Object.fromEntries(controlIds.map((id) => [id, root.querySelector(`#${id}`)]));

  const outputs = Object.fromEntries(
    controlIds.map((id) => [id, root.querySelector(`#${id}-output`)]),
  );

  const averageMetric = root.querySelector('#metric-eff');
  const frontDifferenceMetric = root.querySelector('#metric-df');
  const rearDifferenceMetric = root.querySelector('#metric-dr');
  const statusPanel = root.querySelector('#simulation-status');
  const chart = root.querySelector('#brake-chart');

  const requiredElements = [
    ...Object.values(controls),
    ...Object.values(outputs),
    averageMetric,
    frontDifferenceMetric,
    rearDifferenceMetric,
    statusPanel,
    chart,
  ];

  if (requiredElements.some((element) => !element)) {
    console.warn(
      'O simulador de suspensÃ£o nÃ£o foi inicializado porque existem elementos obrigatÃ³rios ausentes.',
    );

    return undefined;
  }

  let animationFrameId = null;

  const update = () => {
    animationFrameId = null;

    const values = readValues(controls);
    const metrics = calculateSuspensionMetrics(values);

    updateControlOutputs(outputs, values);
    updateMetrics({
      averageMetric,
      frontDifferenceMetric,
      rearDifferenceMetric,
      metrics,
    });

    updateStatus(statusPanel, metrics);
    drawSuspensionChart(chart, values);
  };

  const scheduleUpdate = () => {
    if (animationFrameId !== null) {
      cancelAnimationFrame(animationFrameId);
    }

    animationFrameId = requestAnimationFrame(update);
  };

  const controlListeners = controlIds.map((id) => {
    const listener = () => scheduleUpdate();

    controls[id].addEventListener('input', listener);

    return {
      element: controls[id],
      listener,
    };
  });

  const caseListeners = Array.from(root.querySelectorAll('[data-case]')).map((button) => {
    const listener = () => {
      const selectedCase = cases[button.dataset.case];

      if (!selectedCase) {
        return;
      }

      Object.entries(selectedCase).forEach(([id, value]) => {
        controls[id].value = String(value);
      });

      update();
    };

    button.addEventListener('click', listener);

    return {
      element: button,
      listener,
    };
  });

  update();

  return () => {
    if (animationFrameId !== null) {
      cancelAnimationFrame(animationFrameId);
    }

    controlListeners.forEach(({ element, listener }) => {
      element.removeEventListener('input', listener);
    });

    caseListeners.forEach(({ element, listener }) => {
      element.removeEventListener('click', listener);
    });
  };
}

function readValues(controls) {
  return Object.fromEntries(controlIds.map((id) => [id, Number(controls[id].value)]));
}

function calculateSuspensionMetrics(values) {
  const adhesionValues = [values.fl, values.fr, values.rl, values.rr];

  const average = adhesionValues.reduce((total, value) => total + value, 0) / adhesionValues.length;

  const frontDifference = Math.abs(values.fl - values.fr);
  const rearDifference = Math.abs(values.rl - values.rr);
  const minimumAdhesion = Math.min(...adhesionValues);

  return {
    average,
    frontDifference,
    rearDifference,
    minimumAdhesion,
  };
}

function updateControlOutputs(outputs, values) {
  controlIds.forEach((id) => {
    outputs[id].value = `${formatNumber(values[id], 0)} %`;
  });
}

function updateMetrics({ averageMetric, frontDifferenceMetric, rearDifferenceMetric, metrics }) {
  averageMetric.textContent = `${formatNumber(metrics.average, 1)}%`;

  frontDifferenceMetric.textContent = `${formatNumber(metrics.frontDifference, 0)} p.p.`;

  rearDifferenceMetric.textContent = `${formatNumber(metrics.rearDifference, 0)} p.p.`;
}

function updateStatus(statusPanel, metrics) {
  const classification = classifySuspension(metrics);

  statusPanel.className = `status-panel ${classification.className}`;
  statusPanel.innerHTML = `
    <strong>${classification.title}</strong>
    <p>${classification.description}</p>
  `;
}

function classifySuspension(metrics) {
  const maximumDifference = Math.max(metrics.frontDifference, metrics.rearDifference);

  if (metrics.minimumAdhesion < 40 || maximumDifference > 20) {
    return {
      className: 'critical',
      title: 'CondiÃ§Ã£o didÃ¡tica crÃ­tica',
      description:
        'HÃ¡ Ã­ndice de aderÃªncia muito reduzido ou diferenÃ§a lateral elevada. O veÃ­culo requer investigaÃ§Ã£o tÃ©cnica complementar.',
    };
  }

    return {
      className: 'warning',
      title: 'CondiÃ§Ã£o didÃ¡tica de atenÃ§Ã£o',
      description:
        'Os resultados sugerem desempenho reduzido ou assimetria entre as rodas do mesmo eixo.',
    };
  }

  return {

    description:

  const chartValues = [values.fl, values.fr, values.rl, values.rr];

    const value = index * 20;

        y1="${y}"
        x2="680"
        y2="${y}"

      />

      <text
        x="28"
        y="${y + 5}"
        class="svg-text"

  }).join('');

  const bars = chartValues
    .map((value, index) => {
      const x = 95 + index * 145;

      return `

function readDynamicValues(controls) {
  return {
    mass: Number(controls['dynamic-mass'].value),
    stiffness: Number(controls['dynamic-stiffness'].value),
    damping: Number(controls['dynamic-damping'].value),

function calculateDynamicMetrics(values) {
  const { mass, stiffness, damping, excitationFrequency, roadAmplitude } = values;

  const naturalAngularFrequency = Math.sqrt(stiffness / mass);

  const naturalFrequency = naturalAngularFrequency / (2 * Math.PI);

  const criticalDamping = 2 * Math.sqrt(stiffness * mass);

  const dampingRatio = damping / criticalDamping;

  const dampedFrequency =
    dampingRatio < 1 ? naturalFrequency * Math.sqrt(1 - dampingRatio ** 2) : 0;

  const frequencyRatio = naturalFrequency > 0 ? excitationFrequency / naturalFrequency : 0;

  const denominatorSquared =
    (1 - frequencyRatio ** 2) ** 2 + (2 * dampingRatio * frequencyRatio) ** 2;

  const transmissibility =
    denominatorSquared > 0
      ? Math.sqrt((1 + (2 * dampingRatio * frequencyRatio) ** 2) / denominatorSquared)
      : 0;

  const adhesion = estimateAdhesion({
    dampingRatio,
    frequencyRatio,
    transmissibility,
    roadAmplitude,
  });

  return {
    ...values,
    naturalAngularFrequency,
    naturalFrequency,
    criticalDamping,
    dampingRatio,
    dampedFrequency,
    frequencyRatio,
    transmissibility,
    adhesion,
  };
}

function estimateAdhesion({ dampingRatio, frequencyRatio, transmissibility, roadAmplitude }) {
  const resonancePenalty = 28 * Math.exp(-Math.pow((frequencyRatio - 1) / 0.3, 2));

  const lowDampingPenalty = dampingRatio < 0.2 ? (0.2 - dampingRatio) * 120 : 0;

  const highDampingPenalty = dampingRatio > 0.65 ? (dampingRatio - 0.65) * 25 : 0;

  const transmissibilityPenalty = Math.max(0, transmissibility - 1) * 8;

  const roadPenalty = roadAmplitude * 0.45;

  const adhesion =
    94 -
    resonancePenalty -
    lowDampingPenalty -
    highDampingPenalty -
    transmissibilityPenalty -
    roadPenalty;

  return clamp(adhesion, 20, 95);
}

}

function selectOverallClass(classNames) {
  if (classNames.includes('critical')) {
    return 'critical';
  }

  if (classNames.includes('warning')) {
    return 'warning';
  }

  return 'normal';
}

function clamp(value, minimum, maximum) {
  return Math.min(maximum, Math.max(minimum, value));
}
